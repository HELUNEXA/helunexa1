// Hekimin kendi önerilerini ekleyebilmesi için basit katalog const DOCTOR_CATALOG = [ "Hemogram", "Ferritin", "Açlık glukozu ve HbA1c", "Lipid profili", "TSH", "Serbest T4", "D vitamini · 25-OH D", "Vitamin B12", "Kreatinin / eGFR", "ALT", "AST", "Tam idrar tahlili", "C-Reaktif Protein (CRP)", "Magnezyum", "Çinko", "Demir Bağlama Kapasitesi" ]; window.openDetail = function(id) { currentApp = allApps.find(a => a.id === id); if (!currentApp) return; document.getElementById('modalTitle').textContent = currentApp.fullname; // Hastanın seçtiklerini ayıkla let patientSelectedHtml = ''; let patientTests = []; if (currentApp.test_notes && currentApp.test_notes.includes('Hastanın Hekimle Görüşmek İstediği Tetkikler:')) { let selectedTestsString = currentApp.test_notes.replace('Hastanın Hekimle Görüşmek İstediği Tetkikler: ', ''); patientTests = selectedTestsString.split(' | '); patientSelectedHtml = `
🧪 Hastanın Seçtiği Tetkikler (Talep):
    ${patientTests.map(t => `
  • ${t}
    • `).join('')}
`; } // Hekimin mevcut reçetesi (varsa) let currentPrescription = currentApp.prescription || ''; let appointmentDetail = ''; if (currentApp.appointment_date) { appointmentDetail = `
${new Date(currentApp.appointment_date).toLocaleDateString('tr-TR', { day: '2-digit', month: 'long', year: 'numeric' })} ${currentApp.appointment_time ? ' - Saat: ' + currentApp.appointment_time : ''}
`; } document.getElementById('modalBody').innerHTML = ` ${appointmentDetail} ${patientSelectedHtml}
${currentApp.service_type}
${currentApp.test_notes || '-'}

🩺 Hekim Değerlendirmesi

`; // Hekimin eklediği testleri tutacağımız dizi window.tempDoctorTests = currentPrescription ? currentPrescription.split('\n').filter(t => t.trim() !== '') : []; renderDoctorSelectedList(); document.getElementById('modalOverlay').classList.add('active'); }; // Hekim test eklediğinde çalışacak fonksiyon window.addDoctorTest = function() { const select = document.getElementById('doctorTestSelect'); const test = select.value; if (!test) return; if (!window.tempDoctorTests.includes(test)) { window.tempDoctorTests.push(test); renderDoctorSelectedList(); } select.value = ''; // Seçimi sıfırla }; // Hekim test sildiğinde çalışacak fonksiyon window.removeDoctorTest = function(test) { window.tempDoctorTests = window.tempDoctorTests.filter(t => t !== test); renderDoctorSelectedList(); }; // Eklenen testleri ekrana çizen fonksiyon window.renderDoctorSelectedList = function() { const listContainer = document.getElementById('doctorSelectedList'); const presTextarea = document.getElementById('editPrescription'); if (window.tempDoctorTests.length === 0) { listContainer.innerHTML = 'Henüz tetkik eklenmedi.'; } else { listContainer.innerHTML = window.tempDoctorTests.map(t => ` ${t} `).join(''); } // Textarea'yı bu liste ile senkronize et (Her satıra bir test) presTextarea.value = window.tempDoctorTests.join('\n'); };
